  <script>

    const textA = document.getElementById('text-a');

    const textB = document.getElementById('text-b');

    const countA = document.getElementById('count-a');

    const countB = document.getElementById('count-b');

    const output = document.getElementById('output');

    const modeRadios = document.querySelectorAll('input[name="mode"]');

    function updateCounters() {

      const valA = textA.value;

      const linesA = valA ? valA.split('\n').length : 0;

      countA.textContent = `文字数: ${valA.length} / 行数: ${linesA}`;

      const valB = textB.value;

      const linesB = valB ? valB.split('\n').length : 0;

      countB.textContent = `文字数: ${valB.length} / 行数: ${linesB}`;

    }

    function escapeHtml(str) {

      return str

        .replace(/&/g, '&amp;')

        .replace(/</g, '&lt;')

        .replace(/>/g, '&gt;');

    }

    // 最小編集距離アルゴリズムによる差分計算

    function computeDiff(arrA, arrB) {

      const n = arrA.length;

      const m = arrB.length;

      const dp = Array.from({ length: n + 1 }, () => Array(m + 1).fill(0));

      for (let i = 0; i <= n; i++) dp[i][0] = i;

      for (let j = 0; j <= m; j++) dp[0][j] = j;

      for (let i = 1; i <= n; i++) {

        for (let j = 1; j <= m; j++) {

          if (arrA[i - 1] === arrB[j - 1]) {

            dp[i][j] = dp[i - 1][j - 1];

          } else {

            dp[i][j] = Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1);

          }

        }

      }

      const diff = [];

      let i = n, j = m;

      while (i > 0 || j > 0) {

        if (i > 0 && j > 0 && arrA[i - 1] === arrB[j - 1]) {

          diff.unshift({ type: 'same', value: arrA[i - 1] });

          i--;

          j--;

        } else if (j > 0 && (i === 0 || dp[i][j - 1] <= dp[i - 1][j])) {

          diff.unshift({ type: 'add', value: arrB[j - 1] });

          j--;

        } else if (i > 0 && (j === 0 || dp[i][j - 1] > dp[i - 1][j])) {

          diff.unshift({ type: 'del', value: arrA[i - 1] });

          i--;

        }

      }

      return diff;

    }

    function renderDiff() {

      const mode = document.querySelector('input[name="mode"]:checked').value;

      const valA = textA.value;

      const valB = textB.value;

      if (!valA && !valB) {

        output.innerHTML = '';

        return;

      }

      if (mode === 'line') {

        const linesA = valA.split('\n');

        const linesB = valB.split('\n');

        const diff = computeDiff(linesA, linesB);

        let html = '';

        diff.forEach(item => {

          const esc = escapeHtml(item.value);

          if (item.type === 'add') {

            html += `<span class="diff-line-add">+ ${esc}</span>`;

          } else if (item.type === 'del') {

            html += `<span class="diff-line-del">- ${esc}</span>`;

          } else {

            html += `<span class="diff-line-same">  ${esc}</span>`;

          }

        });

        output.innerHTML = html;

      } else {

        const charsA = Array.from(valA);

        const charsB = Array.from(valB);

        const diff = computeDiff(charsA, charsB);

        let html = '';

        diff.forEach(item => {

          const esc = escapeHtml(item.value);

          if (item.type === 'add') {

            html += `<ins class="diff-added">${esc}</ins>`;

          } else if (item.type === 'del') {

            html += `<del class="diff-removed">${esc}</del>`;

          } else {

            html += esc;

          }

        });

        output.innerHTML = html;

      }

    }

    function handleInput() {

      updateCounters();

      renderDiff();

    }

    textA.addEventListener('input', handleInput);

    textB.addEventListener('input', handleInput);

    modeRadios.forEach(radio => radio.addEventListener('change', renderDiff));

    updateCounters();
